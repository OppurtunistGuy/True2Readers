    70→    70→/**
    71→    71→ * Render a Google Books preview snippet that may contain <em>...</em>
    72→    72→ * highlight tags — without dangerouslySetInnerHTML.
    73→    73→ */
    74→    74→function renderSnippet(snippet: string): React.ReactNode {
    75→    75→  const parts = snippet.split(/(<\/?em>)/)
    76→    76→  const nodes: React.ReactNode[] = []
    77→    77→  let highlight = false
    78→    78→  let key = 0
    79→    79→  for (const part of parts) {
    80→    80→    if (part === '<em>') {
    81→    81→      highlight = true
    82→    82→    } else if (part === '</em>') {
    83→    83→      highlight = false
    84→    84→    } else if (part.length > 0) {
    85→    85→      if (highlight) {
    86→    86→        nodes.push(
    87→    87→          <mark key={key++} className="rounded bg-blue-100 px-0.5 text-blue-900">
    88→    88→            {part}
    89→    89→          </mark>,
    90→    90→        )
    91→    91→      } else {
    92→    92→        nodes.push(<React.Fragment key={key++}>{part}</React.Fragment>)
    93→    93→      }
    94→    94→    }
    95→    95→  }
    96→    96→  return nodes
    97→    97→}
