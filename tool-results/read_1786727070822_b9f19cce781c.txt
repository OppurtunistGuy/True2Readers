    55→/**
    56→ * Render a Google Books preview snippet that may contain <em>...</em>
    57→ * highlight tags — without dangerouslySetInnerHTML.
    58→ */
    59→function renderSnippet(snippet: string): React.ReactNode {
    60→  const parts = snippet.split(/(<\/?em>)/)
    61→  const nodes: React.ReactNode[] = []
    62→  let highlight = false
    63→  let key = 0
    64→  for (const part of parts) {
    65→    if (part === '<em>') {
    66→      highlight = true
    67→    } else if (part === '</em>') {
    68→      highlight = false
    69→    } else if (part.length > 0) {
    70→      if (highlight) {
    71→        nodes.push(
    72→          <mark key={key++} className="rounded bg-blue-100 px-0.5 text-blue-900">
    73→            {part}
    74→          </mark>,
    75→        )
    76→      } else {
    77→        nodes.push(<React.Fragment key={key++}>{part}</React.Fragment>)
    78→      }
    79→    }
    80→  }
    81→  return nodes
    82→}

   320→          {/* Middle: meta + preview */}

   336→                  </span>
   337→                )}
   338→                {readerFit && (
   339→                  <span
   340→                    className="inline-flex items-center gap-1.5 rounded-full bg-slate-100 px-2.5 py-1 text-xs font-medium text-slate-600 ring-1 ring-slate-200"

   356→                    aria-label="Enter reading mode"
