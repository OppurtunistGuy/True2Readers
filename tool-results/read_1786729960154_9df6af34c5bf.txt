    62→/** Build a sliding window of visible page numbers, capped at `maxVisible`. */
    63→function pageWindow(current: number, total: number, maxVisible = 5): number[] {
    64→  if (total <= maxVisible) {
    65→    return Array.from({ length: total }, (_, i) => i + 1)
    66→  }
    67→  const half = Math.floor(maxVisible / 2)
    68→  let start = current - half
    69→  let end = current + half
    70→  if (start < 1) {
    71→    start = 1
    72→    end = maxVisible
    73→  }
    74→  if (end > total) {
    75→    end = total
    76→    start = total - maxVisible + 1
    77→  }
    78→  return Array.from({ length: end - start + 1 }, (_, i) => start + i)
    79→}

    81→/** Derive sorted decade buckets from the current page's books. */
    82→function deriveDecades(books: Book[]): { decade: number; label: string; count: number }[] {
    83→  const counts = new Map<number, number>()
    84→  for (const b of books) {
    85→    if (!b.year) continue
    86→    const y = Number(b.year)
    87→    if (!Number.isFinite(y)) continue
    88→    const decade = Math.floor(y / 10) * 10
    89→    counts.set(decade, (counts.get(decade) ?? 0) + 1)
    90→  }
    91→  return Array.from(counts.entries())
    92→    .map(([decade, count]) => ({ decade, label: `${decade}s`, count }))
    93→    .sort((a, b) => a.decade - b.decade)
    94→}

   104→      className="inline-block font-medium text-slate-700"
   105→    >
   106→      {value.toLocaleString()}
   107→    </motion.span>
   108→  )
   109→}

   318→      </div>
   319→
   320→      {/* Results header */}
   321→      <motion.div
   322→        initial={{ opacity: 0, y: 8 }}
   323→        animate={{ opacity: 1, y: 0 }}
   324→        transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   325→        className="mt-8 flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between"
   326→      >
   327→        <div className="min-w-0">
   328→          <div className="flex items-center gap-2">
   329→            {isAuthor && (
   330→              <span className="inline-flex size-7 items-center justify-center rounded-lg bg-blue-50 text-blue-600">

   334→            <h2

   336→              title={displayQuery}
   337→            >
   338→              {displayQuery}

   371→                compareMode

   374→              )}

   549→            <>
   550→              <span className="ml-2 text-xs font-medium text-slate-500">Rating:</span>
   551→              {RATING_FILTERS.map((rf) => (
   552→                <button
   553→                  key={rf.value}
   554→                  type="button"
   555→                  onClick={() => setMinRating(rf.value)}
   556→                  className={cn(
   557→                    'inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-xs font-medium transition-all active:scale-95 focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-slate-50',
   558→                    minRating === rf.value
   559→                      ? 'bg-blue-600 text-white shadow-sm'
   560→                      : 'border border-slate-200 bg-white text-slate-600 hover:border-blue-200 hover:bg-blue-50 hover:text-blue-700',
   561→                  )}
   562→                  aria-pressed={minRating === rf.value}
   563→                >
   564→                  {rf.value > 0 && <Star className="size-3 fill-current" strokeWidth={0} />}
   565→                  {rf.label}
   566→                </button>
   567→              ))}
   568→            </>
   569→          )}
   570→        </motion.div>
   571→      )}
   572→
   573→      {/* Body */}
   574→      <div className="mt-6">
   575→        {error ? (
   576→          <Alert variant="destructive" className="bg-white">
   577→            <AlertCircle className="size-4" />
   578→            <AlertTitle>Search failed</AlertTitle>
   579→            <AlertDescription>{error}</AlertDescription>
   580→          </Alert>
   581→        ) : isLoading && results.length === 0 ? (
   582→          <div className={cn(
   583→            viewMode === 'grid'
   584→              ? 'grid grid-cols-2 gap-6 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5'
   585→              : 'flex flex-col gap-3',
   586→          )}>
   587→            {Array.from({ length: SKELETON_COUNT }).map((_, i) => (
   588→              viewMode === 'grid' ? (
   589→                <div key={i} className="flex flex-col gap-3">
   590→                  <Skeleton className="trueread-shimmer aspect-[2/3] w-full rounded-lg" />
   591→                  <Skeleton className="trueread-shimmer h-4 w-5/6" />
   592→                  <Skeleton className="trueread-shimmer h-3 w-3/4" />
   593→                  <Skeleton className="trueread-shimmer h-3 w-1/3" />
   594→                </div>
   595→              ) : (
   596→                <div key={i} className="flex items-center gap-4 rounded-xl p-3">
   597→                  <Skeleton className="trueread-shimmer size-16 rounded-lg shrink-0" />
   598→                  <div className="flex-1 flex flex-col gap-2">
   599→                    <Skeleton className="trueread-shimmer h-4 w-2/3" />
   600→                    <Skeleton className="trueread-shimmer h-3 w-1/2" />
   601→                  </div>
   602→                </div>
   603→              )
   604→            ))}
   605→          </div>
   606→        ) : visibleResults.length === 0 ? (
   607→          <EmptyState query={displayQuery} />
   608→        ) : viewMode === 'grid' ? (
   609→          <div
   610→            className={cn(
   611→              'grid grid-cols-2 gap-6 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5',
   612→              isLoading && 'opacity-60 transition-opacity',
   613→            )}
   614→          >
   615→            {visibleResults.map((book, i) => (
   616→              <BookCard
   617→                key={book.id}
   618→                book={book}
   619→                onSelect={onSelectBook}
   620→                index={i}
   621→                variant="grid"
   622→                isBookmarked={bookmarkedIds?.has(book.id) ?? false}
   623→                onToggleBookmark={compareMode ? undefined : onToggleBookmark}
   624→                compareMode={compareMode}
   625→                isCompareSelected={compareSelectedIds.has(book.id)}
   626→                onToggleCompare={handleToggleCompare}
   627→                isFocused={focusedIndex === i}
   628→                mainButtonRef={(el) => {
   629→                  cardRefs.current[i] = el
   630→                }}
   631→              />
   632→            ))}
   633→          </div>
   634→        ) : (
   635→          <div
   636→            className={cn(
   637→              'flex flex-col gap-2 rounded-2xl bg-white p-2 ring-1 ring-slate-200/70 trueread-shadow-card',
   638→              isLoading && 'opacity-60 transition-opacity',
   639→            )}
   640→          >
   641→            {visibleResults.map((book, i) => (
   642→              <BookCard
   643→                key={book.id}
   644→                book={book}
   645→                onSelect={onSelectBook}
   646→                index={i}
   647→                variant="list"
   648→                isBookmarked={bookmarkedIds?.has(book.id) ?? false}
   649→                onToggleBookmark={compareMode ? undefined : onToggleBookmark}
   650→                compareMode={compareMode}
   651→                isCompareSelected={compareSelectedIds.has(book.id)}
   652→                onToggleCompare={handleToggleCompare}
   653→                isFocused={focusedIndex === i}
   654→                mainButtonRef={(el) => {
   655→                  cardRefs.current[i] = el
   656→                }}
   657→              />
   658→            ))}
   659→          </div>
   660→        )}
   661→
   662→        {/* Filter empty state — when filters yield no books. */}
   663→        {!error &&
   664→          !isLoading &&
   665→          visibleResults.length === 0 &&
   666→          sortedResults.length > 0 &&
   667→          (activeDecade != null || minRating > 0) && (
   668→            <div className="mt-6 flex flex-col items-center gap-2 rounded-xl bg-white px-6 py-8 text-center ring-1 ring-slate-200/70">
   669→              <p className="text-sm text-slate-500">
   670→                {activeDecade != null && minRating > 0
   671→                  ? `No books from the ${activeDecade}s with ${minRating}+ stars on this page.`
   672→                  : activeDecade != null
   673→                    ? `No books from the ${activeDecade}s on this page.`
   674→                    : `No books with ${minRating}+ stars on this page.`}
   675→              </p>
   676→              <button
   677→                type="button"
   678→                onClick={() => {
   679→                  setActiveDecade(null)
   680→                  setMinRating(0)
   681→                }}
   682→                className="text-sm font-medium text-blue-600 hover:underline"
   683→              >
   684→                Clear filters
   685→              </button>
   686→            </div>
   687→          )}
   688→      </div>
   689→
   690→      {/* Pagination */}
   691→      {showPagination && !error && sortedResults.length > 0 && (
   692→        <nav
   693→          aria-label="Results pagination"
   694→          className="mt-12 flex flex-wrap items-center justify-center gap-2"
   695→        >
   696→          <Button
   697→            type="button"
   698→            variant="outline"
   699→            size="sm"
   700→            onClick={() => onPageChange(page - 1)}
   701→            disabled={page <= 1 || isLoading}
   702→            className="h-9 gap-1.5 border-slate-200 bg-white px-3 text-slate-600 hover:bg-slate-50 hover:text-slate-900"
   703→          >
   704→            <ArrowLeft className="size-4" />
   705→            <span className="hidden sm:inline">Prev</span>
   706→          </Button>
   707→
   708→          {visiblePages.map((p) => {
   709→            const isActive = p === page
   710→            return (
   711→              <button
   712→                key={p}
   713→                type="button"
   714→                onClick={() => onPageChange(p)}
   715→                disabled={isLoading}
   716→                aria-current={isActive ? 'page' : undefined}
   717→                aria-label={`Go to page ${p}`}
   718→                className={cn(
   719→                  'inline-flex h-9 min-w-9 items-center justify-center rounded-lg px-3 text-sm font-medium transition-all focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-slate-50 active:scale-95',
   720→                  isActive
   721→                    ? 'bg-blue-600 text-white shadow-sm'
   722→                    : 'border border-slate-200 bg-white text-slate-600 hover:bg-slate-50 hover:text-slate-900',
   723→                )}
   724→              >
   725→                {p}
   726→              </button>
   727→            )
   728→          })}
   729→
   730→          <Button
   731→            type="button"
   732→            variant="outline"
   733→            size="sm"
   734→            onClick={() => onPageChange(page + 1)}
   735→            disabled={page >= totalPages || isLoading}
   736→            className="h-9 gap-1.5 border-slate-200 bg-white px-3 text-slate-600 hover:bg-slate-50 hover:text-slate-900"
   737→          >
   738→            <span className="hidden sm:inline">Next</span>
   739→            <ArrowRight className="size-4" />
   740→          </Button>
   741→        </nav>
   742→      )}
   743→
   744→      {/* Compare modal */}
   745→      <CompareModal
   746→        books={compareSelected}
   747→        isOpen={compareOpen}
   748→        onClose={() => setCompareOpen(false)}
   749→        onSelectBook={onSelectBook}
   750→      />
   751→    </div>
   752→  )
   753→}

   761→      className="flex flex-col items-center justify-center gap-3 rounded-2xl bg-white px-6 py-16 text-center ring-1 ring-slate-200/70 trueread-shadow-card"
