    64→/**
    65→ * Render a Google Books preview snippet that may contain <em>...</em>
    66→ * highlight tags — without dangerouslySetInnerHTML.
    67→ */
    68→function renderSnippet(snippet: string): React.ReactNode {
    69→  const parts = snippet.split(/(<\/?em>)/)
    70→  const nodes: React.ReactNode[] = []
    71→  let highlight = false
    72→  let key = 0
    73→  for (const part of parts) {
    74→    if (part === '<em>') {
    75→      highlight = true
    76→    } else if (part === '</em>') {
    77→      highlight = false
    78→    } else if (part.length > 0) {
    79→      if (highlight) {
    80→        nodes.push(
    81→          <mark key={key++} className="rounded bg-blue-100 px-0.5 text-blue-900">
    82→            {part}
    83→          </mark>,
    84→        )
    85→      } else {
    86→        nodes.push(<React.Fragment key={key++}>{part}</React.Fragment>)
    87→      }
    88→    }
    89→  }
    90→  return nodes
    91→}

   529→                    aria-label={`Search more books in ${cat}`}
   530→                  >
   531→                    {cat}
   532→                  </button>
   533→                ))}
   534→                {hiddenCategoryCount > 0 && (
   535→                  <button
   536→                    type="button"
   537→                    onClick={() => setShowAllCategories((v) => !v)}
   538→                    className="inline-flex items-center gap-1 rounded-full bg-blue-50 px-2.5 py-1 text-xs font-medium text-blue-700 transition-all hover:bg-blue-100 active:scale-95"
   539→                    aria-expanded={showAllCategories}
   540→                  >
   541→                    {showAllCategories ? (
   542→                      <>
   543→                        <ChevronUp className="size-3" strokeWidth={2.5} />
   544→                        Show less
   545→                      </>
   546→                    ) : (
   547→                      <>
   548→                        <ChevronDown className="size-3" strokeWidth={2.5} />
   549→                        +{hiddenCategoryCount} more
   550→                      </>
   551→                    )}
   552→                  </button>
   553→                )}
   554→              </div>
   555→            )}
   556→
   557→            {/* Find similar books — primary CTA using the book's first category. */}
   558→            {primaryCategory && (
   559→              <div className="mt-4">
   560→                <button
   561→                  type="button"
   562→                  onClick={() => onSearch(primaryCategory)}
   563→                  className="group/sim inline-flex items-center gap-2 rounded-xl bg-blue-600 px-4 py-2.5 text-sm font-semibold text-white shadow-sm transition-all hover:bg-blue-700 hover:shadow-md active:scale-[0.98] focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-white"
   564→                  aria-label={`Find similar books in ${primaryCategory}`}
   565→                >
   566→                  <Compass className="size-4 transition-transform group-hover/sim:rotate-45" strokeWidth={2.25} />
   567→                  Find similar books
   568→                  <span className="hidden text-blue-200 sm:inline">·</span>
   569→                  <span className="hidden text-blue-100 sm:inline">
   570→                    {primaryCategory}
   571→                  </span>
   572→                </button>
   573→              </div>
   574→            )}
   575→
   576→            <hr className="my-6 border-slate-200" />
   577→
   578→            {/* Preview section */}
   579→            <section aria-labelledby="preview-heading">
   580→              <h2
   581→                id="preview-heading"
   582→                className="flex items-center gap-2 text-sm font-semibold text-blue-600"
   583→              >
   584→                <Quote className="size-4" strokeWidth={2.25} />
   585→                {source === 'google' ? 'Preview (from Google Books)' : 'Preview'}
   586→              </h2>
   587→
   588→              {hasPreview ? (
   589→                <p className="trueread-dropcap mt-3 text-base leading-relaxed text-slate-600">
   590→                  {renderSnippet(book.previewSnippet!)}
   591→                </p>
   592→              ) : (
   593→                <div className="mt-3 flex items-start gap-3 rounded-lg bg-slate-50 p-4 ring-1 ring-slate-200/70">
   594→                  <Info className="mt-0.5 size-5 shrink-0 text-slate-400" />
   595→                  <p className="text-sm text-slate-500">
   596→                    No preview available for this book.
   597→                  </p>
   598→                </div>
   599→              )}
   600→
   601→              {/* Source attribution */}
   602→              <div className="mt-4 flex flex-wrap items-center gap-3 text-xs text-slate-400">
   603→                <span className="inline-flex items-center gap-1">
   604→                  <Globe className="size-3" strokeWidth={2} />
   605→                  Source: {sourceLabel}
   606→                </span>
   607→                {book.previewLink && (
   608→                  <a
   609→                    href={book.previewLink}
   610→                    target="_blank"
   611→                    rel="noopener noreferrer"
   612→                    className="inline-flex items-center gap-1 text-blue-600 transition-colors hover:text-blue-700 hover:underline"
   613→                  >
   614→                    Open in {sourceLabel}
   615→                    <ExternalLink className="size-3" />
   616→                  </a>
   617→                )}
   618→              </div>
   619→            </section>
   620→
   621→            {/* Optional About block — only shown when there's no preview */}
   622→            {showAboutBlock && (
   623→              <section
   624→                aria-labelledby="about-heading"
   625→                className="mt-8 border-t border-slate-200 pt-6"
   626→              >
   627→                <h2
   628→                  id="about-heading"
   629→                  className="flex items-center gap-2 text-sm font-semibold text-slate-900"
   630→                >
   631→                  <BookOpen className="size-4 text-blue-600" strokeWidth={2.25} />
   632→                  About this book
   633→                </h2>
   634→                <p
   635→                  className={cn(
   636→                    'trueread-dropcap mt-3 text-sm leading-relaxed text-slate-600',
   637→                    !showFullDesc && 'line-clamp-4',
   638→                  )}
   639→                >
   640→                  {book.description}
   641→                </p>
   642→                <div className="mt-3 flex items-center gap-3">
   643→                  {book.description && book.description.length > 280 && (
   644→                    <button
   645→                      type="button"
   646→                      onClick={() => setShowFullDesc((v) => !v)}
   647→                      className="inline-flex items-center gap-1 text-xs font-medium text-blue-600 transition-colors hover:text-blue-700 hover:underline"
   648→                    >
   649→                      {showFullDesc ? (
   650→                        <>
   651→                          <ChevronUp className="size-3" strokeWidth={2.5} />
   652→                          Show less
   653→                        </>
   654→                      ) : (
   655→                        <>
   656→                          <ChevronDown className="size-3" strokeWidth={2.5} />
   657→                          Show more
   658→                        </>
   659→                      )}
   660→                    </button>
   661→                  )}
   662→                </div>
   663→              </section>
   664→            )}
   665→
   666→            {/* Explore related genres */}
   667→            {exploreCategories.length > 0 && (
   668→              <section
   669→                aria-label="Explore related topics"
   670→                className="mt-8 border-t border-slate-200 pt-6"
   671→              >
   672→                <div className="mb-3 flex items-center gap-2">
   673→                  <span className="inline-flex size-7 items-center justify-center rounded-lg bg-blue-50 text-blue-600">
   674→                    <Sparkles className="size-3.5" strokeWidth={2.25} />
   675→                  </span>
   676→                  <h3 className="text-sm font-semibold text-slate-900">
   677→                    Explore related
   678→                  </h3>
   679→                </div>
   680→                <div className="flex flex-col gap-1.5">
   681→                  {exploreCategories.map((cat, i) => (
   682→                    <motion.button
   683→                      key={cat}
   684→                      type="button"
   685→                      onClick={() => onSearch(cat)}
   686→                      initial={{ opacity: 0, x: -4 }}
   687→                      animate={{ opacity: 1, x: 0 }}
   688→                      transition={{ duration: 0.2, delay: i * 0.04, ease: [0.22, 1, 0.36, 1] }}
   689→                      className="group/rel flex items-center gap-2.5 rounded-lg px-3 py-2 text-left transition-all hover:bg-blue-50/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 active:scale-[0.98]"
   690→                      aria-label={`Search books in ${cat}`}
   691→                    >
   692→                      <Search className="size-3.5 text-slate-300 transition-colors group-hover/rel:text-blue-400" strokeWidth={2.5} />
   693→                      <span className="flex-1 text-sm font-medium text-slate-600 transition-colors group-hover/rel:text-blue-700">
   694→                        {cat}
   695→                      </span>
   696→                      <ArrowLeft className="size-3 rotate-180 text-slate-300 transition-colors group-hover/rel:text-blue-400" strokeWidth={2.5} />
   697→                    </motion.button>
   698→                  ))}
   699→                </div>
   700→              </section>
   701→            )}
   702→
   703→            {/* AI breakdown — visible inline below the middle column on md.
   704→                On lg+ the grid places it in the right column instead. */}
   705→            <div className="mt-8 lg:hidden">
   706→              <AiBreakdown book={book} />
   707→            </div>
   708→          </div>
   709→
   710→          {/* Right: AI breakdown (lg+ only) */}
   711→          <aside className="hidden lg:block">
   712→            <AiBreakdown book={book} />
   713→          </aside>
   714→        </div>
   715→      </motion.article>
   716→
   717→      {/* More from this author — real author search, excludes current book.
   718→          Renders only when the search yields additional results. */}
   719→      {onSelectBook && (
   720→        <div className="mx-auto mt-2 w-full max-w-6xl px-4 md:px-6">
   721→          <MoreFromAuthor book={book} onSelect={onSelectBook} />
   722→        </div>
   723→      )}
   724→
   725→      {/* Cover lightbox — fullscreen zoom of the book cover */}
   726→      <CoverLightbox
   727→        book={book}
   728→        isOpen={coverLightboxOpen}
   729→        onClose={() => setCoverLightboxOpen(false)}
   730→        source={source}
   731→      />
   732→
   733→      {/* Reading mode overlay */}
   734→      <AnimatePresence>
   735→        {readingMode && readingText && (
   736→          <>
   737→            <motion.div
   738→              initial={{ opacity: 0 }}
   739→              animate={{ opacity: 1 }}
   740→              exit={{ opacity: 0 }}
   741→              transition={{ duration: 0.25 }}
   742→              className="fixed inset-0 z-[60] bg-slate-900/40 backdrop-blur-md"
   743→              onClick={() => setReadingMode(false)}
   744→              aria-hidden="true"
   745→            />
   746→            <motion.div
   747→              initial={{ opacity: 0, y: 20, scale: 0.98 }}
   748→              animate={{ opacity: 1, y: 0, scale: 1 }}
   749→              exit={{ opacity: 0, y: 20, scale: 0.98 }}
   750→              transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   751→              className="fixed inset-x-4 top-4 z-[61] mx-auto max-h-[calc(100vh-2rem)] max-w-3xl overflow-hidden rounded-2xl bg-white shadow-2xl ring-1 ring-slate-200"
   752→              role="dialog"
   753→              aria-modal="true"
   754→              aria-labelledby="reading-mode-title"
   755→            >
   756→              {/* Header */}
   757→              <div className="flex items-center justify-between border-b border-slate-100 px-6 py-4">
   758→                <div className="flex min-w-0 items-center gap-3">
   759→                  <span className="inline-flex size-9 shrink-0 items-center justify-center rounded-xl bg-blue-100 text-blue-600">
   760→                    <BookOpen className="size-5" strokeWidth={2.25} />
   761→                  </span>
   762→                  <div className="min-w-0">
   763→                    <h2
   764→                      id="reading-mode-title"
   765→                      className="truncate text-base font-semibold text-slate-900"
   766→                      title={book.title}
   767→                    >
   768→                      {book.title}
   769→                    </h2>
   770→                    <p className="truncate text-xs text-slate-500">
   771→                      by {authorsText}
   772→                    </p>
   773→                  </div>
   774→                </div>
   775→                <button
   776→                  type="button"
   777→                  onClick={() => setReadingMode(false)}
   778→                  className="inline-flex size-8 shrink-0 items-center justify-center rounded-lg text-slate-400 transition-colors hover:bg-slate-100 hover:text-slate-700"
   779→                  aria-label="Exit reading mode"
   780→                >
   781→                  <Minimize2 className="size-4" strokeWidth={2} />
   782→                </button>
   783→              </div>
   784→
   785→              {/* Body — scrollable reading content */}
   786→              <div className="max-h-[calc(100vh-8rem)] overflow-y-auto px-6 py-8 sm:px-10 sm:py-10">
   787→                <div className="mx-auto max-w-2xl">
   788→                  {/* Reading time chip */}
   789→                  {readingTime && (
   790→                    <div className="mb-6 flex items-center justify-center">
   791→                      <span className="inline-flex items-center gap-1.5 rounded-full bg-blue-50 px-3 py-1 text-xs font-medium text-blue-700 ring-1 ring-blue-100">
   792→                        <Clock className="size-3.5" strokeWidth={2.25} />
   793→                        {readingTime.label} read
   794→                      </span>
   795→                    </div>
   796→                  )}
   797→                  <p className="trueread-dropcap text-lg leading-relaxed text-slate-700 sm:text-xl sm:leading-relaxed">
   798→                    {hasPreview ? renderSnippet(book.previewSnippet!) : book.description}
   799→                  </p>
   800→                </div>
   801→              </div>
   802→
   803→              {/* Footer */}
   804→              <div className="border-t border-slate-100 bg-slate-50/50 px-6 py-3">
   805→                <p className="flex items-center justify-center gap-1.5 text-center text-xs text-slate-400">
   806→                  <Globe className="size-3" strokeWidth={2} />
   807→                  Source: {sourceLabel}
   808→                  <span aria-hidden="true">·</span>
   809→                  Press{' '}
   810→                  <kbd className="rounded border border-slate-200 bg-white px-1 py-0.5 text-[9px] font-semibold text-slate-500">
   811→                    Esc
   812→                  </kbd>{' '}
   813→                  to exit
   814→                </p>
   815→              </div>
   816→            </motion.div>
   817→          </>
   818→        )}
   819→      </AnimatePresence>
   820→    </div>
   821→  )
   822→}
