    50→    50→    50→## API contract (shared — both backend & frontend must conform)
    51→    51→    51→
    52→    52→    52→```ts
    53→    53→    53→// src/components/trueread/types.ts
    54→    54→    54→export interface Book {
    55→    55→    55→  id: string
    56→    56→    56→  title: string
    57→    57→    57→  authors: string[]
    58→    58→    58→  year: string | null            // 4-digit year from publishedDate
    59→    59→    59→  cover?: string                 // https image URL
    60→    60→    60→  description?: string
    61→    61→    61→  previewSnippet?: string        // searchInfo.textSnippet (may contain <em>...</em>)
    62→    62→    62→  previewLink?: string
    63→    63→    63→  categories?: string[]
    64→    64→    64→  publisher?: string
    65→    65→    65→  pageCount?: number | null
    66→    66→    66→  rating?: number | null
    67→    67→    67→  embeddable?: boolean
    68→    68→    68→  viewability?: string
    69→    69→    69→}
    70→    70→    70→
    71→    71→    71→export interface SearchResponse {
    72→    72→    72→  query: string
    73→    73→    73→  mode: 'title' | 'genre'
    74→    74→    74→  bestMatch: Book | null
    75→    75→    75→  results: Book[]
    76→    76→    76→  totalItems: number
    77→    77→    77→  startIndex: number
    78→    78→    78→  maxResults: number
    79→    79→    79→}
    80→    80→    80→```
    81→    81→    81→
    82→    82→    82→Endpoint: `GET /api/books/search?q=<query>&startIndex=<0>&maxResults=<24>` → `SearchResponse`
    83→    83→    83→- On startIndex 0: backend decides mode via title-match heuristic (token Jaccard ≥ 0.6 or substring between query and top result's title). Title match → mode 'title', bestMatch = top result. Else mode 'genre', bestMatch null.
    84→    84→    84→- On startIndex > 0: always mode 'genre' (pagination), bestMatch null.

   124→   124→   124→  - decideMode(query, results): 0 results → {genre, null}; 1 result → {title, results[0]}; >1 → title iff isTitleMatch(query, results[0].title) else {genre, null}.
