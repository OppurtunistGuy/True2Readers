    70→/**
    71→ * Render a Google Books preview snippet that may contain <em>...</em>
    72→ * highlight tags — without dangerouslySetInnerHTML.
    73→ */
    74→function renderSnippet(snippet: string): React.ReactNode {
    75→  const parts = snippet.split(/(<\/?em>)/)
    76→  const nodes: React.ReactNode[] = []
    77→  let highlight = false
    78→  let key = 0
    79→  for (const part of parts) {
    80→    if (part === '<em>') {
    81→      highlight = true
    82→    } else if (part === '</em>') {
    83→      highlight = false
    84→    } else if (part.length > 0) {
    85→      if (highlight) {
    86→        nodes.push(
    87→          <mark key={key++} className="rounded bg-blue-100 px-0.5 text-blue-900">
    88→            {part}
    89→          </mark>,
    90→        )
    91→      } else {
    92→        nodes.push(<React.Fragment key={key++}>{part}</React.Fragment>)
    93→      }
    94→    }
    95→  }
    96→  return nodes
    97→}

   534→                  className="trueread-no-print inline-flex items-center gap-1.5 rounded-full bg-white px-2.5 py-1 text-xs font-medium text-slate-600 ring-1 ring-slate-200 transition-all hover:bg-slate-50 hover:text-slate-800 active:scale-95"
   535→                  aria-label="Share this book"
   536→                >
   537→                  {copied ? (
   538→                    <>
   539→                      <Check className="size-3.5 text-green-600" strokeWidth={2.25} />
   540→                      <span className="text-green-600">Copied!</span>
   541→                    </>
   542→                  ) : (
   543→                    <>
   544→                      <Share2 className="size-3.5" strokeWidth={2.25} />
   545→                      Share
   546→                    </>
   547→                  )}
   548→                </button>
   549→                {/* Bookmark chip with kbd hint — mirrors the cover bookmark button */}
   550→                {onToggleBookmark && (
   551→                  <button
   552→                    type="button"
   553→                    onClick={() => onToggleBookmark(book)}
   554→                    className={cn(
   555→                      'inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs font-medium ring-1 transition-all active:scale-95',
   556→                      isBookmarked
   557→                        ? 'bg-blue-50 text-blue-700 ring-blue-200 hover:bg-blue-100'
   558→                        : 'bg-white text-slate-600 ring-slate-200 hover:bg-blue-50 hover:text-blue-700 hover:ring-blue-200',
   559→                    )}
   560→                    aria-pressed={isBookmarked}
   561→                    aria-label={isBookmarked ? 'Remove from reading list' : 'Add to reading list'}
   562→                  >
   563→                    {isBookmarked ? (
   564→                      <Bookmark className="size-3.5" strokeWidth={2.25} />
   565→                    ) : (
   566→                      <Bookmark className="size-3.5" strokeWidth={2} />
   567→                    )}
   568→                    {isBookmarked ? 'Saved' : 'Save'}
   569→                    <kbd className="ml-0.5 hidden rounded border border-slate-200 bg-slate-50 px-1 text-[9px] font-semibold text-slate-400 sm:inline">
   570→                      B
   571→                    </kbd>
   572→                  </button>
   573→                )}
   574→              </div>
   575→            )}
   576→
   577→            {/* Category badges — clickable to trigger a genre search (collapsible) */}
   578→            {book.categories && book.categories.length > 0 && (
   579→              <div className="mt-3 flex flex-wrap gap-2">
   580→                {visibleCategories.map((cat) => (
   581→                  <button
   582→                    key={cat}
   583→                    type="button"
   584→                    onClick={() => handleCategoryClick(cat)}
   585→                    className="rounded-full bg-slate-100 px-2.5 py-1 text-xs font-medium text-slate-600 transition-all hover:bg-blue-50 hover:text-blue-700 focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-white active:scale-95"
   586→                    aria-label={`Search more books in ${cat}`}
   587→                  >
   588→                    {cat}
   589→                  </button>
   590→                ))}
   591→                {hiddenCategoryCount > 0 && (
   592→                  <button
   593→                    type="button"
   594→                    onClick={() => setShowAllCategories((v) => !v)}
   595→                    className="inline-flex items-center gap-1 rounded-full bg-blue-50 px-2.5 py-1 text-xs font-medium text-blue-700 transition-all hover:bg-blue-100 active:scale-95"
   596→                    aria-expanded={showAllCategories}
   597→                  >
   598→                    {showAllCategories ? (
   599→                      <>
   600→                        <ChevronUp className="size-3" strokeWidth={2.5} />
   601→                        Show less
   602→                      </>
   603→                    ) : (
   604→                      <>
   605→                        <ChevronDown className="size-3" strokeWidth={2.5} />
   606→                        +{hiddenCategoryCount} more
   607→                      </>
   608→                    )}
   609→                  </button>
   610→                )}
   611→              </div>
   612→            )}
   613→
   614→            {/* Find similar books — primary CTA using the book's first category. */}
   615→            {primaryCategory && (
   616→              <div className="mt-4">
   617→                <button
   618→                  type="button"
   619→                  onClick={() => onSearch(primaryCategory)}
   620→                  className="group/sim inline-flex items-center gap-2 rounded-xl bg-blue-600 px-4 py-2.5 text-sm font-semibold text-white shadow-sm transition-all hover:bg-blue-700 hover:shadow-md active:scale-[0.98] focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-white"
   621→                  aria-label={`Find similar books in ${primaryCategory}`}
   622→                >
   623→                  <Compass className="size-4 transition-transform group-hover/sim:rotate-45" strokeWidth={2.25} />
   624→                  Find similar books
   625→                  <span className="hidden text-blue-200 sm:inline">·</span>
   626→                  <span className="hidden text-blue-100 sm:inline">
   627→                    {primaryCategory}
   628→                  </span>
   629→                </button>
   630→              </div>
   631→            )}
   632→
   633→            <hr className="my-6 border-slate-200" />
   634→
   635→            {/* Preview section */}
   636→            <section aria-labelledby="preview-heading">
   637→              <h2
   638→                id="preview-heading"
   639→                className="flex items-center gap-2 text-sm font-semibold text-blue-600"
   640→              >
   641→                <Quote className="size-4" strokeWidth={2.25} />
   642→                {source === 'google' ? 'Preview (from Google Books)' : 'Preview'}
   643→              </h2>
   644→
   645→              {hasPreview ? (
   646→                <p className="trueread-dropcap mt-3 text-base leading-relaxed text-slate-600">
   647→                  {renderSnippet(book.previewSnippet!)}
   648→                </p>
   649→              ) : (
   650→                <div className="mt-3 flex items-start gap-3 rounded-lg bg-slate-50 p-4 ring-1 ring-slate-200/70">
   651→                  <Info className="mt-0.5 size-5 shrink-0 text-slate-400" />
   652→                  <p className="text-sm text-slate-500">
   653→                    No preview available for this book.
   654→                  </p>
   655→                </div>
   656→              )}
   657→
   658→              {/* Source attribution */}
   659→              <div className="mt-4 flex flex-wrap items-center gap-3 text-xs text-slate-400">
   660→                <span className="inline-flex items-center gap-1">
   661→                  <Globe className="size-3" strokeWidth={2} />
   662→                  Source: {sourceLabel}
   663→                </span>
   664→                {book.previewLink && (
   665→                  <a
   666→                    href={book.previewLink}
   667→                    target="_blank"
   668→                    rel="noopener noreferrer"
   669→                    className="inline-flex items-center gap-1 text-blue-600 transition-colors hover:text-blue-700 hover:underline"
   670→                  >
   671→                    Open in {sourceLabel}
   672→                    <ExternalLink className="size-3" />
   673→                  </a>
   674→                )}
   675→              </div>
   676→            </section>
   677→
   678→            {/* Optional About block — only shown when there's no preview */}
   679→            {showAboutBlock && (
   680→              <section
   681→                aria-labelledby="about-heading"
   682→                className="mt-8 border-t border-slate-200 pt-6"
   683→              >
   684→                <h2
   685→                  id="about-heading"
   686→                  className="flex items-center gap-2 text-sm font-semibold text-slate-900"
   687→                >
   688→                  <BookOpen className="size-4 text-blue-600" strokeWidth={2.25} />
   689→                  About this book
   690→                </h2>
   691→                <p
   692→                  className={cn(
   693→                    'trueread-dropcap mt-3 text-sm leading-relaxed text-slate-600',
   694→                    !showFullDesc && 'line-clamp-4',
   695→                  )}
   696→                >
   697→                  {book.description}
   698→                </p>
   699→                <div className="mt-3 flex items-center gap-3">
   700→                  {book.description && book.description.length > 280 && (
   701→                    <button
   702→                      type="button"
   703→                      onClick={() => setShowFullDesc((v) => !v)}
   704→                      className="inline-flex items-center gap-1 text-xs font-medium text-blue-600 transition-colors hover:text-blue-700 hover:underline"
   705→                    >
   706→                      {showFullDesc ? (
   707→                        <>
   708→                          <ChevronUp className="size-3" strokeWidth={2.5} />
   709→                          Show less
   710→                        </>
   711→                      ) : (
   712→                        <>
   713→                          <ChevronDown className="size-3" strokeWidth={2.5} />
   714→                          Show more
   715→                        </>
   716→                      )}
   717→                    </button>
   718→                  )}
   719→                </div>
   720→              </section>
   721→            )}
   722→
   723→            {/* Explore related genres */}
   724→            {exploreCategories.length > 0 && (
   725→              <section
   726→                aria-label="Explore related topics"
   727→                className="mt-8 border-t border-slate-200 pt-6"
   728→              >
   729→                <div className="mb-3 flex items-center gap-2">
   730→                  <span className="inline-flex size-7 items-center justify-center rounded-lg bg-blue-50 text-blue-600">
   731→                    <Sparkles className="size-3.5" strokeWidth={2.25} />
   732→                  </span>
   733→                  <h3 className="text-sm font-semibold text-slate-900">
   734→                    Explore related
   735→                  </h3>
   736→                </div>
   737→                <div className="flex flex-col gap-1.5">
   738→                  {exploreCategories.map((cat, i) => (
   739→                    <motion.button
   740→                      key={cat}
   741→                      type="button"
   742→                      onClick={() => onSearch(cat)}
   743→                      initial={{ opacity: 0, x: -4 }}
   744→                      animate={{ opacity: 1, x: 0 }}
   745→                      transition={{ duration: 0.2, delay: i * 0.04, ease: [0.22, 1, 0.36, 1] }}
   746→                      className="group/rel flex items-center gap-2.5 rounded-lg px-3 py-2 text-left transition-all hover:bg-blue-50/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 active:scale-[0.98]"
   747→                      aria-label={`Search books in ${cat}`}
   748→                    >
   749→                      <Search className="size-3.5 text-slate-300 transition-colors group-hover/rel:text-blue-400" strokeWidth={2.5} />
   750→                      <span className="flex-1 text-sm font-medium text-slate-600 transition-colors group-hover/rel:text-blue-700">
   751→                        {cat}
   752→                      </span>
   753→                      <ArrowLeft className="size-3 rotate-180 text-slate-300 transition-colors group-hover/rel:text-blue-400" strokeWidth={2.5} />
   754→                    </motion.button>
   755→                  ))}
   756→                </div>
   757→              </section>
   758→            )}
   759→
   760→            {/* AI breakdown — visible inline below the middle column on md.
   761→                On lg+ the grid places it in the right column instead. */}
   762→            <div className="mt-8 lg:hidden">
   763→              <AiBreakdown book={book} />
   764→            </div>
   765→          </div>
   766→
   767→          {/* Right: AI breakdown (lg+ only) */}
   768→          <aside className="hidden lg:block">
   769→            <AiBreakdown book={book} />
   770→          </aside>
   771→        </div>
   772→      </motion.article>
   773→
   774→      {/* More from this author — real author search, excludes current book.
   775→          Renders only when the search yields additional results.
   776→          The data attribute lets the print CSS hide this section (it relies on
   777→          a live network fetch, which doesn't belong in a printed summary). */}
   778→      {onSelectBook && (
   779→        <div
   780→          className="mx-auto mt-2 w-full max-w-6xl px-4 md:px-6"
   781→          data-trueread-more-from-author
   782→        >
   783→          <MoreFromAuthor book={book} onSelect={onSelectBook} />
   784→        </div>
   785→      )}
   786→
   787→      {/* Cover lightbox — fullscreen zoom of the book cover */}
   788→      <CoverLightbox
   789→        book={book}
   790→        isOpen={coverLightboxOpen}
   791→        onClose={() => setCoverLightboxOpen(false)}
   792→        source={source}
   793→      />
   794→
   795→      {/* Reading mode overlay */}
   796→      <AnimatePresence>
   797→        {readingMode && readingText && (
   798→          <>
   799→            <motion.div
   800→              initial={{ opacity: 0 }}
   801→              animate={{ opacity: 1 }}
   802→              exit={{ opacity: 0 }}
   803→              transition={{ duration: 0.25 }}
   804→              className="fixed inset-0 z-[60] bg-slate-900/40 backdrop-blur-md"
   805→              onClick={() => setReadingMode(false)}
   806→              aria-hidden="true"
   807→            />
   808→            <motion.div
   809→              initial={{ opacity: 0, y: 20, scale: 0.98 }}
   810→              animate={{ opacity: 1, y: 0, scale: 1 }}
   811→              exit={{ opacity: 0, y: 20, scale: 0.98 }}
   812→              transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   813→              className="fixed inset-x-4 top-4 z-[61] mx-auto max-h-[calc(100vh-2rem)] max-w-3xl overflow-hidden rounded-2xl bg-white shadow-2xl ring-1 ring-slate-200"
   814→              role="dialog"
   815→              aria-modal="true"
   816→              aria-labelledby="reading-mode-title"
   817→            >
   818→              {/* Header */}
   819→              <div className="flex items-center justify-between border-b border-slate-100 px-6 py-4">
   820→                <div className="flex min-w-0 items-center gap-3">
   821→                  <span className="inline-flex size-9 shrink-0 items-center justify-center rounded-xl bg-blue-100 text-blue-600">
   822→                    <BookOpen className="size-5" strokeWidth={2.25} />
   823→                  </span>
   824→                  <div className="min-w-0">
   825→                    <h2
   826→                      id="reading-mode-title"
   827→                      className="truncate text-base font-semibold text-slate-900"
   828→                      title={book.title}
   829→                    >
   830→                      {book.title}
   831→                    </h2>
   832→                    <p className="truncate text-xs text-slate-500">
   833→                      by {authorsText}
   834→                    </p>
   835→                  </div>
   836→                </div>
   837→                <button
   838→                  type="button"
   839→                  onClick={() => setReadingMode(false)}
   840→                  className="inline-flex size-8 shrink-0 items-center justify-center rounded-lg text-slate-400 transition-colors hover:bg-slate-100 hover:text-slate-700"
   841→                  aria-label="Exit reading mode"
   842→                >
   843→                  <Minimize2 className="size-4" strokeWidth={2} />
   844→                </button>
   845→              </div>
   846→
   847→              {/* Body — scrollable reading content */}
   848→              <div className="max-h-[calc(100vh-8rem)] overflow-y-auto px-6 py-8 sm:px-10 sm:py-10">
   849→                <div className="mx-auto max-w-2xl">
   850→                  {/* Reading time chip */}
   851→                  {readingTime && (
   852→                    <div className="mb-6 flex items-center justify-center">
   853→                      <span className="inline-flex items-center gap-1.5 rounded-full bg-blue-50 px-3 py-1 text-xs font-medium text-blue-700 ring-1 ring-blue-100">
   854→                        <Clock className="size-3.5" strokeWidth={2.25} />
   855→                        {readingTime.label} read
   856→                      </span>
   857→                    </div>
   858→                  )}
   859→                  <p className="trueread-dropcap text-lg leading-relaxed text-slate-700 sm:text-xl sm:leading-relaxed">
   860→                    {hasPreview ? renderSnippet(book.previewSnippet!) : book.description}
   861→                  </p>
   862→                </div>
   863→              </div>
   864→
   865→              {/* Footer */}
   866→              <div className="border-t border-slate-100 bg-slate-50/50 px-6 py-3">
   867→                <p className="flex items-center justify-center gap-1.5 text-center text-xs text-slate-400">
   868→                  <Globe className="size-3" strokeWidth={2} />
   869→                  Source: {sourceLabel}
   870→                  <span aria-hidden="true">·</span>
   871→                  Press{' '}
   872→                  <kbd className="rounded border border-slate-200 bg-white px-1 py-0.5 text-[9px] font-semibold text-slate-500">
   873→                    Esc
   874→                  </kbd>{' '}
   875→                  to exit
   876→                </p>
   877→              </div>
   878→            </motion.div>
   879→          </>
   880→        )}
   881→      </AnimatePresence>
   882→    </div>
   883→  )
   884→}
