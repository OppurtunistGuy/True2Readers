    75→/**
    76→ * Render a Google Books preview snippet that may contain <em>...</em>
    77→ * highlight tags — without dangerouslySetInnerHTML.
    78→ */
    79→function renderSnippet(snippet: string): React.ReactNode {
    80→  const parts = snippet.split(/(<\/?em>)/)
    81→  const nodes: React.ReactNode[] = []
    82→  let highlight = false
    83→  let key = 0
    84→  for (const part of parts) {
    85→    if (part === '<em>') {
    86→      highlight = true
    87→    } else if (part === '</em>') {
    88→      highlight = false
    89→    } else if (part.length > 0) {
    90→      if (highlight) {
    91→        nodes.push(
    92→          <mark key={key++} className="rounded bg-blue-100 px-0.5 text-blue-900">
    93→            {part}
    94→          </mark>,
    95→        )
    96→      } else {
    97→        nodes.push(<React.Fragment key={key++}>{part}</React.Fragment>)
    98→      }
    99→    }
   100→  }
   101→  return nodes
   102→}

   677→                  type="button"

   679→                  className="group/sim inline-flex items-center gap-2 rounded-xl bg-blue-600 px-4 py-2.5 text-sm font-semibold text-white shadow-sm transition-all hover:bg-blue-700 hover:shadow-md active:scale-[0.98] focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2 focus-visible:ring-offset-white"

   698→                className="flex items-center gap-2 text-sm font-semibold text-blue-600"

   704→              {hasPreview ? (

   939→                    Esc

   949→      {/* Personalized reading pace calculator — modal where the user picks

   952→      <PaceCalculator

   955→        pageCount={book.pageCount}
   956→        title={book.title}
   957→      />
   958→    </div>
   959→  )
   960→}
